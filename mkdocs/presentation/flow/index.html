  IDX --> D["Router::handle($start)"]
  D --> E["new Request()"]
  E --> F{"handleCors() — Hostname ok?"}
  F -- No --> ERR1["401 → die"]
  F -- Sì --> G{"findMatch() — URL + metodo"}
  G -- No match --> ERR2["404 → die"]
  G -- Metodo errato --> ERR3["405 → die"]
  G -- Match --> H["Route trovata"]
  H --> I{"runMiddleware()"}
  I -- Fallisce --> ERR4["Bloccato → die"]
  I -- Passa tutto --> J["manageRequest()"]
  J -- Eccezione --> ERR5["catch() → 4xx/500"]
  J -- Ok --> K["finally {} — Calcola X-time"]
  ERR5 --> K
  K --> L["sendResponse()"]
  L --> M(["die → risposta al client"])

    $index = 0;
$next = function () use (&$index, $middleware, $req, &$next, $final) {
    if ($index < count($middleware)) {
        require Config::path("directories.middlewares")."/".$middleware[$index].".php";
        $current = new $middleware[$index]();
        $index++;
        $current($req, $next);
    } else {
        $final($req);
    }
};
